43WISE_STATUS wise_timer_init()
44{
45 int i;
46
47 if(timerInited)
48 {
49 return WISE_SUCCESS;
50 }
51
52 memset(timerCallbacks, 0, sizeof(timerCallbacks));
53 timerInited = 1;
54
55 for(i = 0; i < GPTMR_MAX_EVENTS; i++)
56 {
57#ifdef CHIP_TICK_TIMER_CHANNEL
58 if(i == CHIP_TICK_TIMER_CHANNEL)
59 continue;
60#endif
61 hal_intf_gptmr_register_callback(i, _hal_timer_isr, NULL);
62 }
63
64 return WISE_SUCCESS;
65}
66
67WISE_STATUS wise_timer_deinit()
68{
69 int i;
70
71 for(i = 0; i < CHIP_TIMER_CHANNEL_NUM; i++)
72 {
73#ifdef CHIP_TICK_TIMER_CHANNEL
74 if(i == CHIP_TICK_TIMER_CHANNEL)
75 continue;
76#endif
77
78 hal_intf_module_clk_disable(TIMER_MODULES[i]);
79 }
80
81 for(i = 0; i < GPTMR_MAX_EVENTS; i++)
82 {
83#ifdef CHIP_TICK_TIMER_CHANNEL
84 if(i == CHIP_TICK_TIMER_CHANNEL)
85 continue;
86#endif
87 hal_intf_gptmr_unregister_callback(i);
88 }
89
90 timerInited = 0;
91 return WISE_SUCCESS;
92}

102WISE_STATUS wise_timer_config(uint8_t timer_channel, WISE_TIMER_CTRL_T *config)
103{
104 TIMER_TYPE_T timerType;
105
106 if (!IS_VALID_TIMER_CHANNEL(timer_channel)) {
107 return WISE_FAIL;
108 }
109
110#ifdef CHIP_TICK_TIMER_CHANNEL
111 if(timer_channel == CHIP_TICK_TIMER_CHANNEL)
112 {
113 WISE_LOG_ERR("timer ch %d is internal used\n", timer_channel);
114 return WISE_FAIL;
115 }
116#endif
117
118 if (hal_intf_module_clk_is_enabled(TIMER_MODULES[timer_channel]) == DISABLE) {
119 hal_intf_module_clk_enable(TIMER_MODULES[timer_channel]);
120 }
121
122 if(config->start_offset >= WISE_TIMER_MAX_INTERVAL)
123 return WISE_FAIL;
124
125 if(config->interval != WISE_TIMER_COUNTER_MAX)
126 {
127 if(config->interval >= WISE_TIMER_MAX_INTERVAL)
128 {
129 //out of range
130 return WISE_FAIL;
131 }
132 }
133
134 wise_timer_stop(timer_channel);
135
136 if(config->type == WISE_TIMER_PERIODIC)
137 timerType = TIMER_PERIODIC;
138 else
139 timerType = TIMER_ONE_SHOT;
140
141 hal_intf_gptmr_config(timer_channel,
142 timerType,
143 config->interrupt_enable,
144 config->start_offset,
145 config->interval);
146 return WISE_SUCCESS;
147}
