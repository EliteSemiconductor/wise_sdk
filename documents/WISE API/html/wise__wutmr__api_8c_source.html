57uint32_t wise_wutmr_clk_calibrate(uint32_t intervalMs)
58{
59 uint32_t calTargetStart = hal_intf_wutmr_get_counter();
60 uint32_t calSourceStart = hal_intf_sys_tick_get_counter();
61 int progressCount = 1;
62
63 if(intervalMs == 0)
64 intervalMs = 1000;
65
66 WISE_LOG_INFO("WUTMR SW clk calibration");
67 while(1)
68 {
69 uint32_t timelapsed = hal_intf_sys_tick_get_counter() - calSourceStart;
70
71 if(timelapsed > MS_TO_CLK(intervalMs))
72 {
73 wutmrCalClkRate = hal_intf_wutmr_get_counter() - calTargetStart;
74 WISE_LOG_INFO("finish\n");
75 break;
76 }
77
78 if(timelapsed >= progressCount * (MS_TO_CLK(200)))
79 {
80 progressCount++;
81 WISE_LOG_INFO(".\n");
82 }
83
84 __asm("nop");
85 }
86
87 wutmrCalClkRate = wutmrCalClkRate * 1000 / intervalMs;
88 hal_intf_wutmr_set_clock_base(wutmrCalClkRate);
89
90 WISE_LOG_INFO("WUTMR cal: %lu\n", wutmrCalClkRate);
91 return wutmrCalClkRate;
92}
