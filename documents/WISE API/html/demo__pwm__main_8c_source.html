213
214 /* Register callback for oneshot completion */
215 st = wise_pwm_register_callback(PWM_CHANNEL, demo_pwm_callback, NULL);

220 printf("PWM interrupt callback registered\n");
221
222 /* Start PWM */
223 g_pwm_complete = 0;
224 wise_pwm_start(1 << PWM_CHANNEL);
225 printf("\nPWM started - generating %u pulses\n", PWM_ONESHOT_COUNT);
226 printf("Connect oscilloscope to GPIO%u to observe waveform\n\n", PWM_GPIO_PIN);
227
228 /* Wait for completion */
229 printf("Waiting for PWM completion...\n");
230 while (!g_pwm_complete) {
231 __NOP();
232 }
233
234 printf("[PWM%u] ONESHOT_COMPLETE\n", PWM_CHANNEL);
235 printf("\nPWM one-shot sequence complete!\n");
236 printf("Generated %u pulses at %luHz\n", PWM_ONESHOT_COUNT, pwm_conf.u.freq.frequency_Hz);
237
238#endif

241 __WFI();
242 }
