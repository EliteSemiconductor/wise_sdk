52WISE_STATUS wise_aes_exec(WISE_AES_CONFIG_CTX_T *ctx)
53{
54 int ret = WISE_FAIL;
55
56#ifdef CHIP_CRYPTO_HAS_AES
57 hal_intf_aes_mode_and_dir_config(ctx->aes_direction, ctx->aes_mode);
58
59 HAL_AES_DMA_SETUP_T dma_cfg = {.data_io = {.input = ctx->data_cfg.input,
60 .output = ctx->data_cfg.output,
61 .input_length = ctx->data_cfg.length,
62 .output_length = ctx->data_cfg.length},
63 .auth_in = NULL};
64
65 ret = hal_intf_aes_cfg_init_vect(ctx->data_cfg.iv_or_cnt);
66 if (ret != WISE_SUCCESS) {
67 return ret;
68 }
69
70 ret = hal_intf_aes_dma_update(&dma_cfg);
71 if (ret != WISE_SUCCESS) {
72 return ret;
73 }
74 hal_intf_aes_start(ctx->data_cfg.length, AES_DMA_DATA_TRIG);
75 if (ctx->aes_mode == AES_MODE_CBC_MAC) {
76 hal_intf_aes_wait_tag_finished();
77 hal_intf_aes_read_tag((uint32_t *)ctx->data_cfg.output);
78 } else {
79 hal_intf_aes_wait_finished();
80 }
81 return WISE_SUCCESS;
82#else
83 return WISE_FAIL;

98WISE_STATUS wise_aes_ccm_configure(WISE_AES_CCM_CTX_T *ctx)
99{
100#ifdef CHIP_CRYPTO_HAS_AES
101 if (!ctx || (ctx->ccm_cfg.nonce_len > 13) || (ctx->ccm_cfg.tag_len % 2 != 0) || (ctx->ccm_cfg.auth_len > 0 && ctx->ccm_cfg.auth_data == NULL)) {
102 return WISE_FAIL;
103 }
104
105 AES_DIR_T dir = ctx->is_encrypt ? AES_ENCRYPT : AES_DECRYPT;
106 int ret = hal_intf_aes_configure(dir, AES_MODE_CCM, ctx->key_cfg.swap_mode, ctx->key_cfg.key_src, ctx->key_cfg.key_size, ctx->key_cfg.key_bytes,
107 ctx->key_cfg.key_num);
108 if (ret != HAL_NO_ERR) {
109 return WISE_FAIL;
110 }
111
112 if (ctx->ccm_cfg.auth_len > 0) {
113 ctx->auth_cfg.input = ctx->ccm_cfg.auth_data;
114 ctx->auth_cfg.output = NULL;
115 ctx->auth_cfg.input_length = ctx->ccm_cfg.auth_len;
116 ctx->has_auth = true;
117 } else {
118 ctx->has_auth = false;
119 }
120
121 ret = hal_drv_aes_set_ccm_info(ctx->ccm_cfg.nonce, ctx->ccm_cfg.nonce_len, ctx->ccm_cfg.tag_len, ctx->ccm_cfg.auth_len);
122 return (ret == HAL_NO_ERR) ? WISE_SUCCESS : WISE_FAIL;
123#else

127WISE_STATUS wise_aes_ccm_exec(WISE_AES_CCM_CTX_T *ctx, const WISE_AES_DATA_T *data)
128{
129#ifdef CHIP_CRYPTO_HAS_AES
130 int ret = WISE_FAIL;
131
132 if (!ctx || !data || data->input == NULL || data->output == NULL || (!ctx->is_encrypt && data->length < ctx->ccm_cfg.tag_len)) {
133 return WISE_FAIL;
134 }
135
136 uint32_t payload_len = ctx->is_encrypt ? data->length : (data->length - ctx->ccm_cfg.tag_len);
137
138 HAL_AES_DMA_SETUP_T dma_cfg = {
139 .data_io =
140 {
141 .input = data->input,
142 .output = data->output,
143 .input_length = data->length,
144 .output_length = ctx->is_encrypt ? (payload_len + ctx->ccm_cfg.tag_len) : payload_len,
145 },
146 .auth_in = ctx->has_auth ? &ctx->auth_cfg : NULL,
147 };
148
149 //ret = hal_intf_aes_dma_setup(&dma_cfg);
150 ret = hal_intf_aes_dma_update(&dma_cfg);
151 if (ret != WISE_SUCCESS) {
152 return ret;
153 }
154
155 if (ctx->is_encrypt) {
156 if (ctx->ccm_cfg.auth_len > 0) {
157 hal_intf_aes_start(data->length, AES_DMA_AUTH_TRIG);
158 }
159
160 hal_intf_aes_wait_tag_finished();
161
162 if (data->length > 0) {
163 hal_intf_dma_aes_in_channel_trigger();
164 hal_intf_aes_start(data->length, AES_DMA_DATA_TRIG);
165 }
166
167 hal_intf_aes_wait_finished();
168 return WISE_SUCCESS;
169
170 } else {
171 hal_intf_aes_start(data->length, AES_DMA_DATA_TRIG);
172 hal_intf_aes_wait_tag_finished();
173
174 if (data->length != ctx->ccm_cfg.tag_len) {
175 hal_intf_dma_aes_in_channel_trigger();
176 }
177
178 if (ctx->ccm_cfg.auth_len > 0) {
179 hal_intf_aes_start(data->length, AES_DMA_AUTH_TRIG);
180 } else {
181 hal_intf_aes_start(data->length, AES_DMA_DATA_TRIG);
182 }
183
184 hal_intf_aes_wait_finished();
185
186 return (hal_intf_aes_ccm_decryption_result() == WISE_SUCCESS) ? WISE_SUCCESS : WISE_FAIL;
187 }
188#else
189 return WISE_FAIL;
190#endif
191}

193WISE_STATUS wise_sha_generate(E_SHA_MODE_T mode, uint32_t dataAddr, uint32_t dataLen, uint8_t *outputSha)
194{
195#ifdef CHIP_CRYPTO_HAS_SHA
196 uint8_t shaMode;
197
198 if ((!outputSha) || (!dataAddr) || (!dataLen)) {
199 return WISE_FAIL;
200 }
201
202 switch (mode) {
203 case E_SHA_MODE_224:
204 shaMode = (SHA_BIT_224 | SHA_IN_SWAP);
205 break;
206
207 case E_SHA_MODE_256:
208 shaMode = (SHA_BIT_256 | SHA_IN_SWAP);
209 break;
210
211 default:
212 return WISE_FAIL;
213 }
214
215 if (HAL_NO_ERR != hal_intf_sha_generate(shaMode, dataAddr, dataLen, outputSha)) {
216 return WISE_FAIL;
217 }
218
219 return WISE_SUCCESS;
220#else
221 return WISE_FAIL;
222#endif
