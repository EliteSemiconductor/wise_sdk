73WISE_STATUS wise_core_init(void)
74{
75 WISE_STATUS status = WISE_SUCCESS;
76
77 /* Wrong chip: the register map cannot be trusted, so touch neither PMU nor
78 XIP/cache - reprogramming XIP while executing from flash can fault before
79 UART exists. The caller records the failure and still brings UART up. */
80 if(ESMT_SOC_CHIP_ID != wise_sys_get_chip_id()) {
81 return WISE_FAIL;
82 }
83
84 _wise_sys_init();
85 wise_sys_lock(); // for soc remapping
86
87#ifdef CHIP_XIP_SUPPORT_RUNTIME_CONFIG
88 _core_xip_mode_config();
89#endif
90
91 /* Kept here, not in wise_init(): callers that use wise_core_init() alone
92 (radio_ctrl, boot loaders) must also get the cache. */
93 if(WISE_SUCCESS != wise_sys_cache_config(CACHE_SIZE_8K_BYTE)) {
94 status = WISE_FAIL;
95 }
96
97 return status;
98}

100__attribute__((used)) void debugHardfault(uint32_t *sp)
101{
102 uint32_t cfsr = SCB->CFSR;
103 uint32_t hfsr = SCB->HFSR;
104 uint32_t mmfar = SCB->MMFAR;
105 uint32_t bfar = SCB->BFAR;
106
107 uint32_t r0 = sp[0];
108 uint32_t r1 = sp[1];
109 uint32_t r2 = sp[2];
110 uint32_t r3 = sp[3];
111 uint32_t r12 = sp[4];
112 uint32_t lr = sp[5];
113 uint32_t pc = sp[6];
114 uint32_t psr = sp[7];
115
116 debug_print("HardFault:\n");
117 debug_print("SCB->CFSR 0x%08lx\n", cfsr);
118 debug_print("SCB->HFSR 0x%08lx\n", hfsr);
119 debug_print("SCB->MMFAR 0x%08lx\n", mmfar);
120 debug_print("SCB->BFAR 0x%08lx\n", bfar);
121 debug_print("\n");
122
123 debug_print("SP 0x%08lx\n", (uint32_t)sp);
124 debug_print("R0 0x%08lx\n", r0);
125 debug_print("R1 0x%08lx\n", r1);
126 debug_print("R2 0x%08lx\n", r2);
127 debug_print("R3 0x%08lx\n", r3);
128 debug_print("R12 0x%08lx\n", r12);
129 debug_print("LR 0x%08lx\n", lr);
130 debug_print("PC 0x%08lx\n", pc);
131 debug_print("PSR 0x%08lx\n", psr);
132
133 debug_print("\n");
134
135#ifdef _HARDFAULT_FORCE_RESET
136 wise_sys_chip_reset();
137#endif
138
139 while (1);
140}

142#if defined(__CC_ARM)
143/* ARM Compiler 5 does not accept GNU extended inline assembly, so the fault
144 * trampoline is written as an embedded-assembler function. It captures the
145 * stacked frame pointer (MSP or PSP, per EXC_RETURN bit 2) and tail-calls the
146 * C handler debugHardfault(uint32_t *sp). */
147__asm void HardFault_Handler(void)
148{
149 IMPORT debugHardfault
150 TST lr, #4
151 ITE EQ
152 MRSEQ r0, MSP
153 MRSNE r0, PSP
154 LDR r1, =debugHardfault
155 BX r1
156}
157#else
158void HardFault_Handler()
159{
160 __asm volatile("tst lr, #4 \n"
161 "ite eq \n"
162 "mrseq r0, msp \n"
163 "mrsne r0, psp \n"
164 "ldr r1, debugHardfault_address \n"
165 "bx r1 \n"
166 "debugHardfault_address: .word debugHardfault \n");
167
168 while (1);
169}

172void _core_xip_mode_config()
173{
174 WISE_MSBI_INFO_T msbi;
175 uint8_t xipMode, xipClk;
176#ifdef CHIP_XIP_FORCE_SPI_QUAD_40M
177 const bool force_spi_quad_40m = true;
178#else
179 const bool force_spi_quad_40m = false;
180#endif
181
182 _flash_probe();
183
184 if(WISE_SUCCESS == wise_flash_shadow_read_msbi_info(&msbi))
185 {
186 //bootrom has already configured it according to msbi
187 return;
188 }
189
190 if(WISE_SUCCESS == _flash_get_xip_cfg(&xipMode, &xipClk))
191 {
192 if (force_spi_quad_40m) {
193 xipMode = XIP_SPI_MODE_QUAD;
194 xipClk = XIP_SPI_CLK_40M;
195 }
196
197 hal_intf_pmu_set_xip_clk(xipClk);
198 hal_intf_xip_set_spi_mode(xipMode);
199 }
200}
