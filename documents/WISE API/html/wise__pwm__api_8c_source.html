52static WISE_STATUS _do_pwm_ch_configiture(uint8_t channel, WISE_PWM_CONF_T *pwm_conf)
53{
54 HAL_PWM_CONF_T hal_pwm_conf = {0};
55
56 if (pwm_conf->common.mode == PWM_MODE_PERIOD)
57 {
58 hal_pwm_conf.period = pwm_conf->u.period.period;
59 hal_pwm_conf.active_period = pwm_conf->u.period.active_period;
60 hal_pwm_conf.center_align_en = pwm_conf->common.center_align_en;
61 hal_pwm_conf.oneshot_en = pwm_conf->common.oneshot_en;
62 hal_pwm_conf.oneshot_num = pwm_conf->common.oneshot_num;
63 hal_intf_pwm_cfg_output_period(1u << channel, &hal_pwm_conf);
64 }
65 else if (pwm_conf->common.mode == PWM_MODE_FREQUENCY)
66 {
67 hal_pwm_conf.frequency_Hz = pwm_conf->u.freq.frequency_Hz;
68 hal_pwm_conf.duty_percent = pwm_conf->u.freq.duty_percent;
69 hal_pwm_conf.center_align_en = pwm_conf->common.center_align_en;
70 hal_pwm_conf.oneshot_en = pwm_conf->common.oneshot_en;
71 hal_pwm_conf.oneshot_num = pwm_conf->common.oneshot_num;
72 hal_intf_pwm_cfg_output_frequency(1u << channel, &hal_pwm_conf);
73 }
74 else
75 {
76 return WISE_FAIL;
77 }
78
79 if (pwm_conf->common.oneshot_en == ENABLE)
80 {
81 hal_intf_pwm_enable_oneshot_interrupt(1u << channel);
82 }
83 else
84 {
85 hal_intf_pwm_disable_oneshot_interrupt(1u << channel);
86 }
87
88 hal_intf_pwm_set_io_for_channel_pin(channel, &hal_pwm_conf);
89
90 return WISE_SUCCESS;
91}

93WISE_STATUS wise_pwm_configure(uint8_t channel, uint8_t gpio_index, WISE_PWM_CONF_T *pwm_conf)
94{
95 if ((pwm_conf == NULL) || (!PWM_IS_INITED()))
96 {
97 return WISE_FAIL;
98 }
99
100 if(!PWM_IS_CLK_ENABLED())
101 {
102 wise_pmu_module_clk_enable(PWM_MODULE);
103 PWM_SET_CLK_ENABLED();
104 }
105
106 wise_pwm_io_disable_pin(gpio_index);
107
108 _do_pwm_ch_configiture(channel, pwm_conf);
109
110 wise_gpio_set_pwm(channel, gpio_index, ENABLE);
111 wise_gpio_set_mode(gpio_index, MODE_PWM);
112
113 PWM_CH_SET_CONF(channel);
114
115 return WISE_SUCCESS;
116}

129void wise_pwm_start(uint32_t channel_mask)
130{
131 int i;
132
133 for(i = 0; i < CHIP_PWM_CHANNEL_NUM; i++)
134 {
135 if((channel_mask >> i) & 0x1)
136 {
137 if(PWM_CH_IS_CONF(i))
138 PWM_CH_SET_START(i);
139 else
140 channel_mask &= ~(1 << i);
141 }
142 }
143
144 if(channel_mask)
145 hal_intf_pwm_start(channel_mask);
146}
147
148void wise_pwm_stop(uint32_t channel_mask)
149{
150 int i;
151
152 for(i = 0; i < CHIP_PWM_CHANNEL_NUM; i++)
153 {
154 if((channel_mask >> i) & 0x1)
155 {
156 if(PWM_CH_IS_START(i))
157 PWM_CH_CLR_START(i);
158 else
159 channel_mask &= ~(1 << i);
160 }
161 }
162
163 if(channel_mask)
164 hal_intf_pwm_stop(channel_mask);
165}
