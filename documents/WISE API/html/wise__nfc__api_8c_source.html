137static void _clear_data_and_dynamic_lock_bits(void)
138{
139 uint32_t zero_block = 0;
140
141 for (int i = 0; i < NFC_TRIG_BLOCK_IDX; i++) {
142 wise_nfc_block_write_data(i, 1, &zero_block);
143 }
144
145 uint32_t block255_val = 0;
146 wise_nfc_block_read_data(NFC_TRIG_BLOCK_IDX, 1, &block255_val);
147
148 block255_val &= 0xFFFF0000;
149
150 wise_nfc_block_write_data(NFC_TRIG_BLOCK_IDX, 1, &block255_val);
151}

153static int _append_text_info(uint8_t *dest_buf, int start_idx, int max_len, const char *text)
154{
155 uint8_t rec[64] = {0};
156 int rLen = 0;
157
158 rec[rLen++] = 0xD1; // Header (TNF=1, SR=1)
159 rec[rLen++] = 0x01; // Type Length
160 int pLenIdx = rLen++;
161 rec[rLen++] = 'T'; // Type: 'T'
162 rec[rLen++] = 2; // Status (UTF-8, 2-byte lang)
163
164 int strLen = snprintf((char *)&rec[rLen], sizeof(rec) - rLen, "en%s", text);
165 if (strLen < 0 || (rLen + strLen) > sizeof(rec)) {
166 return -1; // Buffer overflow prevention
167 }
168 rLen += strLen;
169 rec[pLenIdx] = (uint8_t)(rLen - 3);
170
171 int total_len = 2 + rLen;
172 if (start_idx + total_len > max_len) {
173 return -1;
174 }
175
176 dest_buf[start_idx] = 0x03; // NDEF Message TLV / Record start
177 dest_buf[start_idx + 1] = (uint8_t)rLen;
178 memcpy(&dest_buf[start_idx + 2], rec, rLen);
179
180 return total_len;
181}

183static int _append_esmt_info(uint8_t *dest_buf, int start_idx, int max_len, const uint8_t *payload, int payload_len)
184{
185 uint8_t rec[64] = {0};
186 int rLen = 0;
187
188 rec[rLen++] = 0xD4; // Header (TNF=4 External, SR=1)
189 rec[rLen++] = 0x04; // Type Length ("ESMT")
190 int pLenIdx = rLen++;
191
192 rec[rLen++] = 'E';
193 rec[rLen++] = 'S';
194 rec[rLen++] = 'M';
195 rec[rLen++] = 'T';
196
197 if (rLen + payload_len > sizeof(rec)) {
198 return -1; // Buffer overflow prevention
199 }
200 memcpy(&rec[rLen], payload, payload_len);
201 rLen += payload_len;
202
203 rec[pLenIdx] = (uint8_t)(rLen - 3);
204
205 int total_len = 2 + rLen;
206 if (start_idx + total_len > max_len) {
207 return -1;
208 }
209
210 dest_buf[start_idx] = 0x03;
211 dest_buf[start_idx + 1] = (uint8_t)rLen;
212 memcpy(&dest_buf[start_idx + 2], rec, rLen);
213
214 return total_len;
215}

217void wise_nfc_crtl_tag_config(uint8_t *str)
218{
219 int idx = 0;
220 int totalDataLen = 0;
221 uint32_t tag_cnt_buf[NFC_DATA_BLOCK_IDX] = {0};
222 uint8_t ndefBytes[NFC_DATA_BLOCK_IDX * 4] = {0};
223 int max_ndef_len = sizeof(ndefBytes);
224
225 WISE_NFC_CFG_T nfc_cfg = {
226 .work_mode = 1,
227 .lock_mode = 1,
228 .int_idx = NFC_INT_IDX_1,
229 };
230
231 wise_nfc_config(&nfc_cfg);
232
233 // Append Text Record
234 int written = _append_text_info(ndefBytes, idx, max_ndef_len, (char *)str);
235 if (written < 0) return; // Error handling
236 idx += written;
237
238 // Custom Payload Configuration
239 uint8_t custom_payload[] = {
240 NFC_DATA_BLOCK_IDX,
241 NFC_DATA_BLOCK_LEN,
242 NFC_ACK_BLOCK_IDX,
243 NFC_ACK_BLOCK_LEN,
244 NFC_TRIG_BLOCK_IDX,
245 (uint8_t)(NFC_TRIG_BLOCK_VAL >> 0),
246 (uint8_t)(NFC_TRIG_BLOCK_VAL >> 8),
247 (uint8_t)(NFC_TRIG_BLOCK_VAL >> 16),
248 (uint8_t)(NFC_TRIG_BLOCK_VAL >> 24),
249 };
250
251 // Append ESMT Record
252 written = _append_esmt_info(ndefBytes, idx, max_ndef_len, custom_payload, sizeof(custom_payload));
253 if (written < 0) return; // Error handling
254 idx += written;
255
256 // Terminating Terminator TLV
257 if (idx < max_ndef_len) {
258 ndefBytes[idx++] = 0xFE;
259 }
260 totalDataLen = idx;
261
262 // Preset Header Blocks
263 tag_cnt_buf[0] = 0x48743480;
264 tag_cnt_buf[1] = 0x801CE3E1;
265 tag_cnt_buf[2] = 0x0000009E;
266 tag_cnt_buf[3] = 0x007C10E1;
267
268 // Pack bytes into 32-bit words (Little-Endian assumption based on original code)
269 for (int i = 0; i < totalDataLen; i += 4) {
270 int blockIdx = 4 + (i / 4);
271 if (blockIdx >= NFC_DATA_BLOCK_IDX) break;
272
273 uint32_t word = 0;
274 int remaining = totalDataLen - i;
275 int copy_len = (remaining < 4) ? remaining : 4;
276
277 for (int b = 0; b < copy_len; b++) {
278 word |= ((uint32_t)ndefBytes[i + b] << (b * 8));
279 }
280 tag_cnt_buf[blockIdx] = word;
281 }
282
283 wise_nfc_block_write_data(0, NFC_DATA_BLOCK_IDX, tag_cnt_buf);
284}

286int wise_nfc_ctrl_read_data(uint8_t *buf)
287{
288 uint32_t header_block = 0;
289 wise_nfc_block_read_data(NFC_DATA_BLOCK_IDX, 1, &header_block);
290
291 uint8_t *p_bytes = (uint8_t *)&header_block;
292
293 uint16_t len = (p_bytes[0] << 8) | p_bytes[1];
294
295 int total_bytes_to_read = len + 2;
296 int blocks_to_read = (total_bytes_to_read + 3) / 4;
297
298 int dest_idx = 0;
299 for (int i = 2; i < 4 && i < total_bytes_to_read; i++) {
300 buf[dest_idx++] = p_bytes[i];
301 }
302
303 if (blocks_to_read > 1) {
304 wise_nfc_block_read_data(NFC_DATA_BLOCK_IDX + 1, blocks_to_read - 1, (uint32_t *)&buf[dest_idx]);
305 }
306
307 return len;
308}
