14#ifdef CHIP_GPIO_HAS_PIO
15static void _set_all_pins_default()
16{
17 int i;
18
19 for (i = 0; i < CHIP_GPIO_NUM; i++) {
20 if ((i == 10) || (i == 14) || (i == 15)) { //leave TCXO, SWDIO, SWDCLK as default
21 continue;
22 }
23
24 hal_intf_gpio_set_mode(i, MODE_GPIO);
25 hal_intf_gpio_set_io_dirction(i, GPIO_DIR_INPUT);
26 hal_intf_gpio_set_pull_sel(i, PUSEL_PULL_UP);
27 }
28}

31void wise_gpio_init(void)
32{
33 int i;
34
35 if(_gpio_inited)
36 return;
37
38 for (i = 0; i < CHIP_GPIO_NUM; i++) {
39 pinFuncBackup[i] = MODE_PIO_FUNC_GPIO;
40 }
41
42 hal_intf_module_clk_enable(GPIO_MODULE);
43
44#ifdef CHIP_GPIO_HAS_PIO
45 _set_all_pins_default();
46#endif
47
48 _gpio_inited = 1;
49}

51void wise_gpio_deinit(void)
52{
53#ifdef CHIP_GPIO_HAS_PIO
54 _set_all_pins_default();
55#endif
56 hal_intf_module_clk_disable(GPIO_MODULE);
57 _gpio_inited = 0;
58}

60int32_t wise_gpio_cfg(WISE_GPIO_CFG_T *gpio_cfg)
61{
62 WISE_STATUS status = WISE_SUCCESS;
63 uint8_t pin_idx = gpio_cfg->pin_idx;
64
65 if (pin_idx >= CHIP_GPIO_NUM) {
66 status = WISE_FAIL;
67 goto error;
68 }
69
70 hal_intf_gpio_set_mode(pin_idx, gpio_cfg->mode);
71
72 if (gpio_cfg->mode == MODE_GPIO) // kevinyang, 20241222, GPIO mode only
73 {
74 hal_intf_gpio_set_io_dirction(pin_idx, gpio_cfg->direction);
75 hal_intf_gpio_set_intrpt(pin_idx, gpio_cfg->int_en, gpio_cfg->int_type);
76 }
77
78error:
79 return status;
80}

98{
99 const uint8_t FUNC_MAP[][3] = {
100 //type channel func
101 {PIO_TYPE_UART, 0, PIO_FUNC_UART_TX}, //MODE_PIO_FUNC_UART0_TX

111 {PIO_TYPE_UART, 2, PIO_FUNC_UART_RTS}, //MODE_PIO_FUNC_UART2_RTS
112 {PIO_TYPE_UART, 2, PIO_FUNC_UART_CTS}, //MODE_PIO_FUNC_UART2_CTS
113 {PIO_TYPE_SPI, 0, PIO_FUNC_SPI_CLK}, //MODE_PIO_FUNC_SPI0_CLK
114 {PIO_TYPE_SPI, 0, PIO_FUNC_SPI_CS}, //MODE_PIO_FUNC_SPI0_CS
115 {PIO_TYPE_SPI, 0, PIO_FUNC_SPI_MOSI}, //MODE_PIO_FUNC_SPI0_MOSI
116 {PIO_TYPE_SPI, 0, PIO_FUNC_SPI_MISO}, //MODE_PIO_FUNC_SPI0_MISO
117 {PIO_TYPE_SPI, 0, PIO_FUNC_SPI_HOLD}, //MODE_PIO_FUNC_SPI0_HOLD
118 {PIO_TYPE_SPI, 0, PIO_FUNC_SPI_WP}, //MODE_PIO_FUNC_SPI0_WP
119 {PIO_TYPE_SPI, 1, PIO_FUNC_SPI_CLK}, //MODE_PIO_FUNC_SPI1_CLK
120 {PIO_TYPE_SPI, 1, PIO_FUNC_SPI_CS}, //MODE_PIO_FUNC_SPI1_CS
121 {PIO_TYPE_SPI, 1, PIO_FUNC_SPI_MOSI}, //MODE_PIO_FUNC_SPI1_MOSI
122 {PIO_TYPE_SPI, 1, PIO_FUNC_SPI_MISO}, //MODE_PIO_FUNC_SPI1_MISO
123 {PIO_TYPE_SPI, 1, PIO_FUNC_SPI_HOLD}, //MODE_PIO_FUNC_SPI1_HOLD
124 {PIO_TYPE_SPI, 1, PIO_FUNC_SPI_WP}, //MODE_PIO_FUNC_SPI1_WP
125 {PIO_TYPE_I2C, 0, PIO_FUNC_I2C_SCL}, //MODE_PIO_FUNC_I2C0_SCL
126 {PIO_TYPE_I2C, 0, PIO_FUNC_I2C_SDA}, //MODE_PIO_FUNC_I2C0_SDA
127 {PIO_TYPE_I2C, 1, PIO_FUNC_I2C_SCL}, //MODE_PIO_FUNC_I2C1_SCL
128 {PIO_TYPE_I2C, 1, PIO_FUNC_I2C_SDA}, //MODE_PIO_FUNC_I2C1_SDA
129 };
130 WISE_STATUS status = WISE_FAIL;
131 uint8_t presetIdx = 0;
132
133#ifdef CHIP_GPIO_HAS_PIO
134 if (pinIdx >= CHIP_GPIO_NUM) {
135 goto finish;
136 }
137
138 if (pinFunc == MODE_PIO_FUNC_GPIO) {
139 if (pinFuncBackup[pinIdx] != MODE_PIO_FUNC_GPIO) {
140 presetIdx = pinFuncBackup[pinIdx] - 1;
141 hal_intf_pio_config(FUNC_MAP[presetIdx][0], FUNC_MAP[presetIdx][1], FUNC_MAP[presetIdx][2], PIO_FUNC_DISABLE, pinIdx);
142 }
143
144 hal_intf_gpio_set_mode(pinIdx, MODE_GPIO);
145 } else if (pinFunc < MODE_PIO_FUNC_UNKOWN) {
146 presetIdx = pinFunc - 1;
147
148 if (HAL_NO_ERR != hal_intf_pio_config(FUNC_MAP[presetIdx][0], FUNC_MAP[presetIdx][1], FUNC_MAP[presetIdx][2], PIO_FUNC_ENABLE, pinIdx)) {
149 goto finish;
150 }
151
152 hal_intf_gpio_set_mode(pinIdx, MODE_PIO);
153 } else {
154 goto finish;
155 }
156
157 pinFuncBackup[pinIdx] = pinFunc;
158
159 status = WISE_SUCCESS;
160#endif
161
162finish:
163 return status;
164}

353void wise_gpio_toggle(uint8_t pin_idx)
354{
355 if (wise_gpio_get_direction(pin_idx) == GPIO_DIR_OUTPUT) {
356 GPIO_STATUS pinSts = wise_gpio_read(pin_idx);
357 wise_gpio_write(pin_idx, (pinSts == GPIO_HIGH) ? GPIO_LOW : GPIO_HIGH);
358 } else {
359 WISE_LOG_ERR("GPIO %d is not configured as output!\n", pin_idx);
360 }
