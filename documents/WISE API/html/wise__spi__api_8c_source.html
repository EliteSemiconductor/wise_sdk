25/* Common gate for every transfer entry point: channel valid, opened, and
26 * opened in the role this function requires. Mismatched role returns
27 * WISE_FAIL instead of poking the hardware. */
28static WISE_STATUS _spi_check_opened(uint8_t spi_channel)
29{
30 if (!IS_VALID_SPI_CHANNEL(spi_channel)) {
31 return WISE_INVALID_INDEX;
32 }
33 if (spi_ctrl_array[spi_channel].state != E_SPI_STATE_ON) {
34 WISE_LOG_ERR("SPI ch%d is not enabled\n", spi_channel);
35 return WISE_FAIL;
36 }
37 return WISE_SUCCESS;
38}
39
40static WISE_STATUS _spi_check_role(uint8_t spi_channel, uint8_t expect_role)
41{
42 WISE_STATUS status = _spi_check_opened(spi_channel);
43
44 if (status != WISE_SUCCESS) {
45 return status;
46 }
47 if (spi_ctrl_array[spi_channel].role != expect_role) {
48 WISE_LOG_ERR("SPI ch%d role mismatch (expect %d)\n", spi_channel, expect_role);
49 return WISE_FAIL;
50 }
51 return WISE_SUCCESS;
52}

54static WISE_STATUS _spi_open(uint8_t spi_channel, const WISE_SPI_CONF_T *spi_conf, uint8_t role)
55{
56 WISE_SPI_CTRL_T *spi = &spi_ctrl_array[spi_channel];
57
58 if (!IS_VALID_SPI_CHANNEL(spi_channel) || (!IS_VALID_SPI_CLOCK(spi_conf->clock_sel))) {
59 return WISE_INVALID_INDEX;
60 }
61
62 /* Slave mode supports SINGLE (and 3-wire) IO only. Dual/quad slave RECEIVE
63 * is unreliable on this controller (multi-lane slave RX premature-ends /
64 * drops leading bytes -- HW errata), so reject it at open. */
65 if (role == E_SPI_ROLE_SLAVE &&
66 (spi_conf->io_mode == WISE_SPI_IO_DUAL || spi_conf->io_mode == WISE_SPI_IO_QUAD)) {
67 return WISE_FAIL;
68 }
69
70 if (hal_intf_module_clk_is_enabled(GET_SPI_MODULE(spi_channel)) == DISABLE) {
71 hal_intf_module_clk_enable(GET_SPI_MODULE(spi_channel));
72 }
73
74 spi->state = E_SPI_STATE_ON;
75 spi->role = role;
76
77 spi->cfg.clock_mode = spi_conf->clock_mode;
78 spi->cfg.role = role;
79 spi->cfg.bus_clock = SPI_CLOCK[spi_conf->clock_sel];
80 spi->cfg.data_bit_width = spi_conf->data_bit_width;
81 spi->cfg.data_merge = spi_conf->data_merge;
82 spi->cfg.bit_order = spi_conf->bit_order;
83 spi->cfg.io_mode = (uint8_t)spi_conf->io_mode;
84 spi->cfg.block_mode = spi_conf->block_mode;
85 spi->cfg.dma_enable = spi_conf->dma_enable;
86
87 return hal_intf_spi_set_config(spi_channel, &spi->cfg);
88}

134WISE_STATUS wise_spi_msg_xfer(uint8_t spi_channel, WISE_SPI_XFER_MSG_T *msg)
135{
136 HAL_SPI_TRANS_FMT_T fmt = {0};
137 /* Role-agnostic: usable by master transfers and by slave RX arming
138 * (slave listen path). Only requires the channel to be open. */
139 WISE_STATUS status = _spi_check_opened(spi_channel);
140
141 if (status != WISE_SUCCESS) {
142 return status;
143 }
144
145 fmt.role = spi_ctrl_array[spi_channel].role;
146 fmt.trans_mode = (uint8_t)msg->trans_mode;
147 fmt.tx_unit_count = msg->tx_unit_count;
148 fmt.rx_unit_count = msg->rx_unit_count;
149 fmt.cmd_en = (msg->cmd_en == WISE_SPI_ENABLE) ? 1 : 0;
150 fmt.cmd_value = msg->cmd;
151 fmt.addr_len = (uint8_t)msg->addr_len; /* 0 = no address phase */
152 fmt.addr_fmt = (uint8_t)msg->addr_fmt;
153 fmt.addr_value = msg->address;
154 fmt.dummy_len = msg->dummy_len;
155
156 return hal_intf_spi_xfer_exec(spi_channel, &fmt, msg->tx_data_buff, msg->rx_data_buff);
157}

159WISE_STATUS wise_spi_write(uint8_t spi_channel, void *data_ptr, uint16_t len)
160{
161 WISE_SPI_XFER_MSG_T trans_msg = {0};
162 WISE_STATUS status = _spi_check_role(spi_channel, E_SPI_ROLE_MASTER);
163
164 if (status != WISE_SUCCESS) {
165 return status;
166 }
167 trans_msg.tx_unit_count = len;
168 trans_msg.tx_data_buff = data_ptr;
169 trans_msg.trans_mode = WISE_SPI_TM_WRITE_ONLY;
170
171 return wise_spi_msg_xfer(spi_channel, &trans_msg);
172}
