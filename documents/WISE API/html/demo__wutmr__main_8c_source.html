120static int cmd_start(int argc, char **argv)
121{
122 uint32_t intervalMs;
123
124 if (argc != 2) {
125 printf("Usage: start <interval_ms>\r\n");
126 printf(" Start periodic WUTMR interrupt, press Ctrl+C to stop\r\n");
127 return -1;
128 }
129
130 intervalMs = strtoul(argv[1], NULL, 10);
131 if (intervalMs < 5) {
132 intervalMs = 5;
133 }
134
135 wutmrIntCount = 0; /* Reset interrupt counter */
136 wise_wutmr_enable_int();
137 wise_wutmr_start(intervalMs, WISE_WUTMR_PERIODIC);
138
139 printf("Periodic WUTMR started (interval=%lums), press Ctrl+C to stop...\r\n",
140 (unsigned long)intervalMs);
141
142 while (!shell_poll_break()) {
143 /* Wait for WUTMR interrupt or Ctrl+C */
144 }
145
146 wise_wutmr_stop();
147 printf("Stopped\r\n");
148
149 return 0;
150}

323static int cmd_loop_idle(int argc, char **argv)
324{
325 uint32_t intervalMs;
326 uint32_t count;
327 uint32_t i;
328
329 if (argc != 3) {
330 printf("Usage: loop_idle <interval_ms> <count>\r\n");
331 printf(" Enter CPU idle mode repeatedly for current measurement\r\n");
332 printf(" Example: loop_idle 1000 10 (idle 1s, repeat 10 times)\r\n");
333 return -1;
334 }
335
336 intervalMs = strtoul(argv[1], NULL, 10);
337 if (intervalMs < 5) {
338 intervalMs = 5;
339 }
340
341 count = strtoul(argv[2], NULL, 10);
342 if (count == 0) {
343 count = 1;
344 }
345
346 printf("CPU idle loop started (interval=%lums, count=%lu)\r\n",
347 (unsigned long)intervalMs, (unsigned long)count);
348
349 for (i = 0; i < count; i++) {
350 wise_system_idle(intervalMs);
351 printf("[%lu/%lu] wakeup\r\n", (unsigned long)(i + 1), (unsigned long)count);
352 }
353
354 printf("CPU idle loop completed\r\n");
355
356 return 0;
357}
