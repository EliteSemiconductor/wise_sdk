185static void demo_spi_io_config(uint8_t spi_channel,
186 SPI_IO_WIDTH_T spi_mode)
187{
188 int num_pins = (spi_mode == SPI_IOW_QUAD) ? 6 :
189 (spi_mode == SPI_IOW_3WIRE) ? 3 : 4;
190
191 bool need_tcxo = false;
192 bool need_swd = false;
193
194 if (spi_channel > 1) {
195 printf("[SPI] invalid channel: %u\n", spi_channel);
196 return;
197 }
198
199 for (int i = 0; i < num_pins; i++) {
200 uint8_t pin = spi_pio_cfg[spi_channel][i][1];
201 if (pin == 10) {
202 need_tcxo = true;
203 }
204 if (pin == 14 || pin == 15) {
205 need_swd = true;
206 }
207 }
208
209 if (need_swd) {
210 wise_sys_swd_config(false);
211 }
212 if (need_tcxo) {
213 wise_sys_tcxo_config(false);
214 }
215
216 for (int i = 0; i < num_pins; i++) {
217 wise_gpio_func_cfg(spi_pio_cfg[spi_channel][i][1],
218 spi_pio_cfg[spi_channel][i][0]);
219 }
220}

234static void demo_slave_listen_once(uint16_t recv_count)
235{
236 WISE_SPI_XFER_MSG_T msg = {0};
237
238 memset(rx_buf, 0, sizeof(rx_buf));
239
240 msg.trans_mode = WISE_SPI_TM_READ_ONLY;
241 msg.dummy_len = 1;
242 msg.rx_data_buff = rx_buf;
243 msg.rx_unit_count = recv_count;
244
245 g_xfer_done = 0;
246
247 wise_spi_reset(SPI_CHANNEL, WISE_SPI_RESET_RX);
248 wise_spi_msg_xfer(SPI_CHANNEL, &msg);
249 wise_spi_slave_set_ready(SPI_CHANNEL);
250}

261static void spi_xfer_and_wait(uint8_t ch,
262 WISE_SPI_XFER_MSG_T *msg)
263{
264 g_xfer_done = WISE_FAIL;
265 wise_spi_msg_xfer(ch, msg);
266 while (g_xfer_done == WISE_FAIL) {
267 __NOP();
268 }
269}

288int main(void)
289{
290 WISE_STATUS st;
291
292 demo_app_common_init(DEMO_APP_NAME);
293
294 printf("\n=== SPI Demo (%s) ===\n",
295#if (DEMO_SPI_ROLE == DEMO_SPI_ROLE_MASTER)
296 "MASTER"
297#else
298 "SLAVE"
299#endif
300 );
301
302 demo_spi_io_config(SPI_CHANNEL, SPI_MODE_TYPE);
303
304 st = wise_spi_init();
305 if (st != WISE_SUCCESS) {
306 printf("[ERR] wise_spi_init failed\n");
307 return 0;
308 }
309
310 wise_spi_register_event_callback(SPI_CHANNEL,
311 demo_spi_callback,
312 NULL);
313
314 WISE_SPI_CONF_T cfg = {0};
315 cfg.clock_mode = CLOCK_MODE0;
316 cfg.io_mode = (SPI_MODE_TYPE == SPI_IOW_QUAD) ? WISE_SPI_IO_QUAD :
317 (SPI_MODE_TYPE == SPI_IOW_DUAL) ? WISE_SPI_IO_DUAL : WISE_SPI_IO_SINGLE;
318 cfg.data_bit_width = SPI_DATA_BIT_WIDTH;
319 cfg.clock_sel = SPI_CLOCK_SEL;
320 cfg.bit_order = SPI_MSB_FIRST;
321 cfg.data_merge = WISE_SPI_ENABLE;
322 cfg.block_mode = E_SPI_NONBLOCK_MODE;
323 cfg.dma_enable = WISE_SPI_DISABLE;
324
325#if (DEMO_SPI_ROLE == DEMO_SPI_ROLE_MASTER)
326
327 st = wise_spi_master_open(SPI_CHANNEL, &cfg);
328 if (st != WISE_SUCCESS) {
329 printf("[ERR] master_open failed\n");
330 return 0;
331 }
332
333 for (uint32_t i = 0; i < TEST_DATA_SIZE; i++) {
334 tx_buf[i] = (uint8_t)(i + 1);
335 }
336
337 WISE_SPI_XFER_MSG_T msg = {0};
338 msg.cmd_en = WISE_SPI_ENABLE;
339 msg.trans_mode = WISE_SPI_TM_DMY_WRITE;
340 msg.tx_data_buff = tx_buf;
341 msg.tx_unit_count = TEST_DATA_SIZE;
342
343 if (SPI_MODE_TYPE == SPI_IOW_QUAD) {
344 msg.cmd = WRITE_DATA_QUAD_IO;
345 msg.dummy_len = 4;
346 } else if (SPI_MODE_TYPE == SPI_IOW_DUAL) {
347 msg.cmd = WRITE_DATA_DUAL_IO;
348 msg.dummy_len = 1;
349 } else {
350 msg.cmd = WRITE_DATA_SINGLE_IO;
351 msg.dummy_len = 1;
352 }
353
354 spi_xfer_and_wait(SPI_CHANNEL, &msg);
355
356 printf("Master TX: ");
357 for (uint32_t i = 0; i < TEST_DATA_SIZE; i++) {
358 printf("%02X ", tx_buf[i]);
359 }
360 printf("\n");
361
362#else /* SLAVE */
363
364 st = wise_spi_slave_open(SPI_CHANNEL, &cfg);
365 if (st != WISE_SUCCESS) {
366 printf("[ERR] slave_open failed\n");
367 return 0;
368 }
369
370 printf("Slave listening...\n");
371 demo_slave_listen_once(TEST_DATA_SIZE);
372
373 while (!g_xfer_done) {
374 __NOP();
375 }
376
377 printf("Slave RX: ");
378 for (uint32_t i = 0; i < TEST_DATA_SIZE; i++) {
379 printf("%02X ", rx_buf[i]);
380 }
381 printf("\n");
382
383#endif
384
385 while (1) {
386 __WFI();
387 }
388}
