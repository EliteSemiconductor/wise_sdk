166int main(void)
167{
168 WISE_STATUS st;
169
170 printf("\n=== PWMSLOW Demo (%s) ===\n",
171#if (DEMO_PWMSLOW_MODE == DEMO_PWMSLOW_MODE_CONTINUOUS)
172 "CONTINUOUS"
173#else
174 "ONESHOT"
175#endif
176 );
177
178 demo_app_common_init(DEMO_APP_NAME);
179
180 /* CRITICAL: Configure LFOSC before PWMSLOW initialization */
181 demo_lfosc_config();
182 printf("\n");
183
184 /* Initialize PWMSLOW module */
185 st = wise_pwmslow_init();
186 if (st != WISE_SUCCESS) {
187 printf("[ERR] wise_pwmslow_init failed\n");
188 return 0;
189 }
190 printf("PWMSLOW initialized\n");
191
192 /* Configure PWMSLOW */
193 WISE_PWMSLOW_CONF_T pwmslow_conf;
194 memset(&pwmslow_conf, 0, sizeof(pwmslow_conf));
195
196#if (DEMO_PWMSLOW_MODE == DEMO_PWMSLOW_MODE_CONTINUOUS)
197
198 /* ========== Continuous Mode ========== */
199 pwmslow_conf.common.mode = PWMSLOW_MODE_FREQUENCY;
200 pwmslow_conf.u.freq.frequency_Hz = PWMSLOW_FREQUENCY_HZ;
201 pwmslow_conf.u.freq.duty_percent = PWMSLOW_DUTY_PERCENT;
202 pwmslow_conf.common.idle_status = 0;
203 pwmslow_conf.common.low_active_en = DISABLE;
204 pwmslow_conf.common.center_align_en = DISABLE;
205 pwmslow_conf.common.oneshot_en = DISABLE;
206 pwmslow_conf.common.oneshot_num = 0;
207
208 st = wise_pwmslow_configure(PWMSLOW_GPIO_PIN, &pwmslow_conf);
209 if (st != WISE_SUCCESS) {
210 printf("[ERR] PWMSLOW configure failed\n");
211 return 0;
212 }
213
214 printf("PWMSLOW configured:\n");
215 printf(" GPIO Pin: %u\n", PWMSLOW_GPIO_PIN);
216 printf(" Frequency: %lu Hz\n", pwmslow_conf.u.freq.frequency_Hz);
217 printf(" Duty Cycle: %lu%%\n", pwmslow_conf.u.freq.duty_percent);
218 printf(" Mode: Continuous\n");
219
220 /* Start PWMSLOW */
221 wise_pwmslow_start();
222 printf("\nPWMSLOW started - running continuously\n");
223 printf("Connect oscilloscope to GPIO%u to observe signal\n", PWMSLOW_GPIO_PIN);
224 printf("Expected: %luHz signal, %lu%% duty cycle\n",
225 pwmslow_conf.u.freq.frequency_Hz,
226 pwmslow_conf.u.freq.duty_percent);
227 printf("Note: output frequency depends on LFOSC configuration and calibration\n\n");
228
229#else /* ONESHOT */
230
231 /* ========== One-shot Mode ========== */
232 pwmslow_conf.common.mode = PWMSLOW_MODE_FREQUENCY;
233 pwmslow_conf.u.freq.frequency_Hz = PWMSLOW_FREQUENCY_HZ;
234 pwmslow_conf.u.freq.duty_percent = PWMSLOW_DUTY_PERCENT;
235 pwmslow_conf.common.idle_status = 0;
236 pwmslow_conf.common.low_active_en = DISABLE;
237 pwmslow_conf.common.center_align_en = DISABLE;
238 pwmslow_conf.common.oneshot_en = ENABLE;
239 pwmslow_conf.common.oneshot_num = PWMSLOW_ONESHOT_COUNT;
240
241 st = wise_pwmslow_configure(PWMSLOW_GPIO_PIN, &pwmslow_conf);
242 if (st != WISE_SUCCESS) {
243 printf("[ERR] PWMSLOW configure failed\n");
244 return 0;
245 }
246
247 printf("PWMSLOW configured:\n");
248 printf(" GPIO Pin: %u\n", PWMSLOW_GPIO_PIN);
249 printf(" Frequency: %lu Hz\n", pwmslow_conf.u.freq.frequency_Hz);
250 printf(" Duty Cycle: %lu%%\n", pwmslow_conf.u.freq.duty_percent);
251 printf(" Mode: One-shot\n");
252 printf(" Pulse Count: %u\n", pwmslow_conf.common.oneshot_num);
253
254 /* Register callback for oneshot completion */
255 st = wise_pwmslow_register_callback(PWMSLOW_EVENT_TRIGGER, demo_pwmslow_callback, NULL);
256 if (st != WISE_SUCCESS) {
257 printf("[ERR] PWMSLOW callback registration failed\n");
258 return 0;
259 }
260 printf("PWMSLOW interrupt callback registered\n");
261
262 /* Start PWMSLOW */
263 g_pwmslow_complete = 0;
264 wise_pwmslow_start();
265 printf("\nPWMSLOW started - generating %u pulses\n", PWMSLOW_ONESHOT_COUNT);
266 printf("Connect oscilloscope to GPIO%u to observe signal\n", PWMSLOW_GPIO_PIN);
267 printf("Note: output frequency depends on LFOSC configuration and calibration\n\n");
268
269 /* Wait for completion */
270 printf("Waiting for PWMSLOW completion...\n");
271 while (!g_pwmslow_complete) {
272 __NOP();
273 }
274
275 printf("[PWMSLOW] ONESHOT_COMPLETE\n");
276 printf("\nPWMSLOW one-shot sequence complete!\n");
277 printf("Generated %u pulses at %luHz\n",
278 PWMSLOW_ONESHOT_COUNT,
279 pwmslow_conf.u.freq.frequency_Hz);
280
281#endif
282
283 while (1) {
284 __WFI();
285 }
286}
