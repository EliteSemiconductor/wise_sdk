22 configure

24WISE_STATUS wise_pwmslow_configure(uint8_t gpio_index, WISE_PWMSLOW_CONF_T *pwmslow_conf)
25{
26 if (pwmslow_conf == NULL) {
27 return WISE_INVALID_INDEX;
28 }
29
30 wise_pmu_module_clk_enable(PWM_SLOW_MODULE);
31 wise_pwm_io_disable_pin(gpio_index);
32 HAL_PWMSLOW_CONF_T hal_pwmslow_conf;
33 memset(&hal_pwmslow_conf, 0, sizeof(hal_pwmslow_conf));
34
35 if (pwmslow_conf->common.mode == PWMSLOW_MODE_PERIOD) {
36 hal_pwmslow_conf.period = pwmslow_conf->u.period.period;
37 hal_pwmslow_conf.active_period = pwmslow_conf->u.period.active_period;
38 hal_pwmslow_conf.center_align_en = pwmslow_conf->common.center_align_en;
39 hal_pwmslow_conf.oneshot_en = pwmslow_conf->common.oneshot_en;
40 hal_pwmslow_conf.oneshot_num = pwmslow_conf->common.oneshot_num;
41 hal_intf_pwmslow_cfg_output_period(&hal_pwmslow_conf);
42 } else if (pwmslow_conf->common.mode == PWMSLOW_MODE_FREQUENCY) {
43 hal_pwmslow_conf.frequency_Hz = pwmslow_conf->u.freq.frequency_Hz;
44 hal_pwmslow_conf.duty_percent = pwmslow_conf->u.freq.duty_percent;
45 hal_pwmslow_conf.center_align_en = pwmslow_conf->common.center_align_en;
46 hal_pwmslow_conf.oneshot_en = pwmslow_conf->common.oneshot_en;
47 hal_pwmslow_conf.oneshot_num = pwmslow_conf->common.oneshot_num;
48 hal_intf_pwmslow_cfg_output_period(&hal_pwmslow_conf);
49 hal_intf_pwmslow_cfg_output_frequency(&hal_pwmslow_conf);
50 } else {
51 return WISE_INVALID_INDEX;
52 }
53
54 if (pwmslow_conf->common.oneshot_en == ENABLE) {
55 hal_intf_pwmslow_enable_oneshot_interrupt();
56 } else {
57 hal_intf_pwmslow_disable_oneshot_interrupt();
58 }
59
60 hal_intf_pwmslow_set_io_pin(&hal_pwmslow_conf);
61 wise_gpio_set_pwmslow(gpio_index, ENABLE);
62 wise_gpio_set_mode(gpio_index, MODE_PWM);
63 return WISE_SUCCESS;
64}
