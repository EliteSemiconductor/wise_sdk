11WISE_STATUS wise_rtc_init(void)
12{
13 /* Enable the RTC module clock only. The LFOSC source is a board-level
14 * choice already applied at boot by _platform_init(); this function used to
15 * override it with a fixed internal-16K config, which also powered down the
16 * external 32K XO. Use wise_sys_lfosc_clk_src_config() to change it. */
17 return hal_intf_module_clk_enable(RTC_MODULE);
18}
19
20WISE_STATUS wise_rtc_deinit(void)
21{
22 return hal_intf_module_clk_disable(RTC_MODULE);
23}

25WISE_STATUS wise_rtc_get_time(WISE_RTC_CNT_T *cur_time)
26{
27 if (!cur_time) {
28 return WISE_FAIL;
29 }
30
31 HAL_RTC_CNT_T hal_time;
32 if (hal_intf_rtc_get_time(&hal_time) == WISE_FAIL) {
33 return WISE_FAIL;
34 }
35
36 cur_time->day = hal_time.day;
37 cur_time->time.hour = hal_time.time.hour;
38 cur_time->time.min = hal_time.time.min;
39 cur_time->time.sec = hal_time.time.sec;
40
41 return WISE_SUCCESS;
42}

44WISE_STATUS wise_rtc_set_time(WISE_RTC_CNT_T *cur_time)
45{
46 if (!cur_time) {
47 return WISE_FAIL;
48 }
49
50 /* Out-of-range values cause unexpected behavior (ATCRTC100 DS085 V1.2), so
51 * reject them instead of letting the register mask truncate silently. */
52 if ((cur_time->day > WISE_RTC_DAY_MAX) ||
53 (cur_time->time.hour > WISE_RTC_HOUR_MAX) ||
54 (cur_time->time.min > WISE_RTC_MIN_MAX) ||
55 (cur_time->time.sec > WISE_RTC_SEC_MAX)) {
56 WISE_LOG_ERR("rtc set_time out of range (day %u/%u, %u:%u:%u)\n",
57 cur_time->day, WISE_RTC_DAY_MAX, cur_time->time.hour,
58 cur_time->time.min, cur_time->time.sec);
59 return WISE_FAIL;
60 }
61
62 HAL_RTC_CNT_T hal_time = {.day = cur_time->day,
63 .time = {
64 .hour = cur_time->time.hour,
65 .min = cur_time->time.min,
66 .sec = cur_time->time.sec,
67 }};
68
69 return hal_intf_rtc_set_time(&hal_time);
70}

72WISE_STATUS wise_rtc_set_alarm(WISE_RTC_ALM_CFG_T *alarm_cfg)
73{
74 if (!alarm_cfg) {
75 return WISE_FAIL;
76 }
77
78 /* Same range rule as wise_rtc_set_time(); the alarm has no Day field. */
79 if ((alarm_cfg->time.hour > WISE_RTC_HOUR_MAX) ||
80 (alarm_cfg->time.min > WISE_RTC_MIN_MAX) ||
81 (alarm_cfg->time.sec > WISE_RTC_SEC_MAX)) {
82 WISE_LOG_ERR("rtc set_alarm out of range (%u:%u:%u)\n",
83 alarm_cfg->time.hour, alarm_cfg->time.min, alarm_cfg->time.sec);
84 return WISE_FAIL;
85 }
86
87 HAL_RTC_ALM_CFG_T hal_alarm_cfg = {.channel = alarm_cfg->channel,
88 .enable = alarm_cfg->enable,
89 .int_en = alarm_cfg->int_en,
90 .time = {
91 .hour = alarm_cfg->time.hour,
92 .min = alarm_cfg->time.min,
93 .sec = alarm_cfg->time.sec,
94 }};
95
96 return hal_intf_rtc_set_alarm(&hal_alarm_cfg);
97}
