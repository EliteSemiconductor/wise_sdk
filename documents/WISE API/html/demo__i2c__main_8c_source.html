141{

167static void demo_i2c_io_config(uint8_t i2c_channel)
168{
169 bool need_tcxo = false;
170 bool need_swd = false;
171
172 if (i2c_channel > 1) {
173 printf("[I2C] invalid channel: %u\n", i2c_channel);
174 return;
175 }
176
177 /* Check if special pins are used */
178 for (int i = 0; i < 2; i++) {
179 uint8_t pin = i2c_pio_cfg[i2c_channel][i][1];
180 if (pin == 10) {
181 need_tcxo = true;
182 }
183 if (pin == 14 || pin == 15) {
184 need_swd = true;
185 }
186 }
187
188 /* Disable SWD/TCXO if needed */
189 if (need_swd) {
190 wise_sys_swd_config(false);
191 }
192 if (need_tcxo) {
193 wise_sys_tcxo_config(false);
194 }
195
196 /* Configure pins */
197 for (int i = 0; i < 2; i++) {
198 wise_gpio_func_cfg(i2c_pio_cfg[i2c_channel][i][1],
199 i2c_pio_cfg[i2c_channel][i][0]);
200 }
201}

260int main(void)
261{
262 WISE_STATUS st;
263
264 printf("\n=== I2C Demo (%s) ===\n",
265#if (DEMO_I2C_ROLE == DEMO_I2C_ROLE_MASTER)
266 "MASTER"
267#else
268 "SLAVE"
269#endif
270 );
271
272 demo_app_common_init(DEMO_APP_NAME);
273
274 demo_i2c_io_config(I2C_CHANNEL);
275
276 st = wise_i2c_init();
277 if (st != WISE_SUCCESS) {
278 printf("[ERR] wise_i2c_init failed\n");
279 return 0;
280 }
281
282 st = wise_i2c_open(I2C_CHANNEL);
283 if (st != WISE_SUCCESS) {
284 printf("[ERR] wise_i2c_open failed\n");
285 return 0;
286 }
287
288 wise_i2c_register_event_callback(I2C_CHANNEL, demo_i2c_callback, NULL);
289
290 WISE_I2C_CONF_T cfg = {0};
291 cfg.i2c_idx = I2C_CHANNEL;
292 cfg.i2cEn = I2C_ENABLE;
293 cfg.dmaEn = I2C_DMA_ENABLE;
294 cfg.speedMode = I2C_SPEED_MODE;
295 cfg.bus_hz = I2C_BUS_HZ;
296 cfg.addressing = I2C_ADDRESSING;
297 cfg.target_address = I2C_SLAVE_ADDR;
298
299#if (DEMO_I2C_ROLE == DEMO_I2C_ROLE_MASTER)
300
301 cfg.role = I2C_MASTER;
302 cfg.dir = 1; /* Transmitter */
303
304 st = wise_i2c_config(I2C_CHANNEL, &cfg);
305 if (st != WISE_SUCCESS) {
306 printf("[ERR] master config failed\n");
307 return 0;
308 }
309
310 printf("Master initialized (addr=0x%02X, speed=%s)\n",
311 I2C_SLAVE_ADDR,
312 I2C_SPEED_MODE == 0 ? "100kHz" : (I2C_SPEED_MODE == 1 ? "400kHz" : "1MHz"));
313
314 /* Prepare test data */
315 for (uint32_t i = 0; i < TEST_DATA_SIZE; i++) {
316 tx_buf[i] = (uint8_t)(i + 1);
317 }
318
319 g_xfer_done = 0;
320
321 st = wise_i2c_send_nbyte(I2C_CHANNEL, tx_buf, TEST_DATA_SIZE);
322 if (st != WISE_SUCCESS) {
323 printf("[ERR] transmission failed\n");
324 return 0;
325 }
326
327 printf("Master sending data...\n");
328 i2c_xfer_and_wait();
329
330 printf("[I2C%u] TRANSFER_DONE\n", I2C_CHANNEL);
331
332 printf("Master TX: ");
333 for (uint32_t i = 0; i < TEST_DATA_SIZE; i++) {
334 printf("%02X ", tx_buf[i]);
335 }
336 printf("\n");
337 printf("Transmission complete\n");
338
339#else /* SLAVE */
340
341 cfg.role = I2C_SLAVE;
342 cfg.dir = 0; /* Receiver */
343
344 st = wise_i2c_config(I2C_CHANNEL, &cfg);
345 if (st != WISE_SUCCESS) {
346 printf("[ERR] slave config failed\n");
347 return 0;
348 }
349
350 printf("Slave initialized (addr=0x%02X, speed=%s)\n",
351 I2C_SLAVE_ADDR,
352 I2C_SPEED_MODE == 0 ? "100kHz" : (I2C_SPEED_MODE == 1 ? "400kHz" : "1MHz"));
353
354 /* Pre-register TX response buffer so master-read gets valid data */
355 for (uint32_t i = 0; i < TEST_DATA_SIZE; i++) {
356 slave_tx_buf[i] = (uint8_t)(0xA0 + i);
357 }
358 wise_i2c_slave_set_tx_buf(I2C_CHANNEL, slave_tx_buf, TEST_DATA_SIZE);
359
360 /* Enable bus-stuck watchdog */
361 wise_i2c_slave_set_timeout(I2C_CHANNEL, SLAVE_WATCHDOG_MS);
362
363 demo_slave_listen_once(TEST_DATA_SIZE);
364
365 printf("Slave listening...\n");
366
367 while (1) {
368 /* Watchdog: auto-recover if bus gets stuck */
369 if (wise_i2c_slave_proc(I2C_CHANNEL) == 1) {
370 printf("[I2C%u] watchdog recovery — re-arming\n", I2C_CHANNEL);
371 demo_slave_listen_once(TEST_DATA_SIZE);
372 printf("Slave listening...\n");
373 continue;
374 }
375
376 if (!g_xfer_done) {
377 __NOP();
378 continue;
379 }
380
381 printf("[I2C%u] TRANSFER_DONE\n", I2C_CHANNEL);
382
383 uint8_t rx_count = wise_i2c_slave_get_rx_count(I2C_CHANNEL);
384 if (rx_count > 0) {
385 printf("Slave RX (%u bytes): ", rx_count);
386 for (uint32_t i = 0; i < rx_count; i++) {
387 printf("%02X ", rx_buf[i]);
388 }
389 printf("\n");
390 } else {
391 printf("Slave TX completed (master read)\n");
392 }
393
394 /* Re-arm for next transaction */
395 demo_slave_listen_once(TEST_DATA_SIZE);
396 printf("Slave listening...\n");
397 }
398
399#endif
400
401 while (1) {
402 __WFI();
403 }
404}
