150int32_t wise_sys_lfosc_clk_src_config(WISE_LFOSC_SRC_T clk_cfg)
151{
152#ifdef CHIP_HAS_LFOSC
153 uint32_t clockRate = 32768;
154
155 if (clk_cfg.clk_src > SYS_LFOSC_CLK_SRC_EXTERNAL_32K) {
156 return WISE_FAIL;
157 }
158 if ((clk_cfg.clk_src == SYS_LFOSC_CLK_SRC_INTERNAL_32K) && (clk_cfg.mode.mode_32k > LFOSC_32K_MODE_8K)) {
159 return WISE_FAIL;
160 }
161 if ((clk_cfg.clk_src == SYS_LFOSC_CLK_SRC_INTERNAL_16K) && (clk_cfg.mode.mode_16k > LFOSC_16K_MODE_LP_VOLT_0P6)) {
162 return WISE_FAIL;
163 }
164
165 if (clk_cfg.clk_src == SYS_LFOSC_CLK_DISABLE) {
166 hal_intf_pmu_lfosc_clk_disable();
167 } else if (clk_cfg.clk_src <= SYS_LFOSC_CLK_SRC_EXTERNAL_32K) {
168 hal_intf_sys_switch_sclk_src(clk_cfg.clk_src);
169
170 switch (clk_cfg.clk_src) {
171 case SYS_LFOSC_CLK_SRC_INTERNAL_32K: {
172 switch (clk_cfg.mode.mode_32k) {
173 case LFOSC_32K_MODE_32K:
174 clockRate = 32768;
175 break;
176
177 case LFOSC_32K_MODE_16K:
178 clockRate = 16384;
179 break;
180
181 case LFOSC_32K_MODE_8K:
182 clockRate = 8192;
183 break;
184 }
185 } break;
186
187 case SYS_LFOSC_CLK_SRC_INTERNAL_16K:
188 clockRate = 16384;
189 break;
190
191 case SYS_LFOSC_CLK_SRC_EXTERNAL_32K:
192 clockRate = 32768;
193 break;
194 }
195
196 hal_intf_wutmr_set_clock_base(clockRate);
197 }
198
199 memcpy(&lfoscConfig, &clk_cfg, sizeof(WISE_LFOSC_SRC_T));
200 lfoscConfig.calFinish = 0;
201
202 return WISE_SUCCESS;
203#else
204 return WISE_FAIL;
205#endif
206}

471static WISE_STATUS _asaradc_exec_once(uint32_t timeoutMs)
472{
473 WISE_STATUS result = WISE_SUCCESS;
474 uint32_t startTime = 0;
475
476 hal_intf_asaradc_set_interrupt(ENABLE);
477 hal_intf_asaradc_start();
478
479 startTime = wise_tick_get_counter();
480 while (1) {
481 if (hal_intf_asaradc_is_ready()) {
482 result = WISE_SUCCESS;
483 break;
484 }
485
486 if (((uint32_t)(wise_tick_get_counter() - startTime)) > MS_TO_CLK(timeoutMs)) {
487 result = WISE_FAIL;
488 break;
489 }
490 }
491
492 hal_intf_asaradc_set_interrupt(DISABLE);
493
494 return result;
495}

638static void sort_by_code(WISE_ASARADC_PT_T *p, uint8_t n)
639{
640 for (uint8_t i = 1; i < n; ++i) {
641 uint8_t j = i;
642 while (j > 0 && p[j - 1].code > p[j].code) {
643 swap_pts(&p[j - 1], &p[j]);
644 --j;
645 }
646 }

651static uint8_t find_segment(const WISE_ASARADC_CALIB_T *c, uint16_t code)
652{
653 for (uint8_t i = 0; i < c->nsegs; ++i) {
654 if (code >= c->segs[i].x1 && code <= c->segs[i].x2) {
655 return i;
656 }
657 }
658 return 0xFF;
659}
660
668int asaradc_calib_init(WISE_ASARADC_CALIB_T *ctx, const WISE_ASARADC_PT_T *in, uint8_t n, bool clamp)
669{
670 if (!ctx || !in || n < 2 || n > ASARADC_MAX_POINTS) {
671 return -1; /* invalid args */
672 }
673
674 /* copy & sort */
675 for (uint8_t i = 0; i < n; ++i) {
676 ctx->pts[i] = in[i];
677 }
678 sort_by_code(ctx->pts, n);
679
680 /* build segments */
681 for (uint8_t i = 0; i < n - 1; ++i) {
682 if (ctx->pts[i + 1].code == ctx->pts[i].code) {
683 return -2; /* duplicate code detected */
684 }
685
686 int32_t dy_mV = (int32_t)ctx->pts[i + 1].mv - (int32_t)ctx->pts[i].mv;
687 int32_t dx = (int32_t)ctx->pts[i + 1].code - (int32_t)ctx->pts[i].code;
688 if (dx == 0) {
689 return -3; /* zero-length segment */
690 }
691
692 /* Input voltage(uV) = slope * code + offset */
693 int32_t slope_uV = (dy_mV * 1000) / dx; /* uV/LSB */
694 int32_t offset_uV = (int32_t)ctx->pts[i].mv * 1000 - slope_uV * (int32_t)ctx->pts[i].code;
695
696 ctx->segs[i].x1 = ctx->pts[i].code;
697 ctx->segs[i].x2 = ctx->pts[i + 1].code;
698 ctx->segs[i].slope_uV_per_LSB = slope_uV;
699 ctx->segs[i].offset_uV = offset_uV;
700 }
701
702 ctx->npts = n;
703 ctx->nsegs = n - 1;
704 ctx->clamp = clamp;
705 return 0;
706}
707
716bool wise_asaradc_cal_table_set(const uint16_t *code, const int32_t *mv, uint8_t n, bool clamp)
717{
718 if (!code || !mv || n < 2 || n > ASARADC_MAX_POINTS) {
719 return false;
720 }
721
722 WISE_ASARADC_PT_T tmp[ASARADC_MAX_POINTS];
723 for (uint8_t i = 0; i < n; ++i) {
724 tmp[i].code = (uint16_t)code[i];
725 tmp[i].mv = (int32_t)mv[i];
726 }
727
728 int rc = asaradc_calib_init(&g_calib_ctx, tmp, (uint8_t)n, clamp);
729 return (rc == 0);
730}

737static int32_t asaradc_code_to_mV(const WISE_ASARADC_CALIB_T *ctx, uint16_t code)
738{
739 if (!ctx || ctx->nsegs == 0) {
740 return 0;
741 }
742
743 uint8_t idx = find_segment(ctx, code);
744 const WISE_ASARADC_SEG_T *seg = NULL;
745
746 if (idx == 0xFF) {
747 if (ctx->clamp) {
748 /* Clamp to endpoints (return endpoint mV directly) */
749 if (code < ctx->pts[0].code) {
750 int32_t uV = (int32_t)ctx->pts[0].mv * 1000;
751 return (uV + 500) / 1000;
752 } else {
753 int32_t uV = (int32_t)ctx->pts[ctx->npts - 1].mv * 1000;
754 return (uV + 500) / 1000;
755 }
756 } else {
757 /* Extrapolate: use the first or last segment coefficients */
758 seg = (code < ctx->segs[0].x1) ? &ctx->segs[0] : &ctx->segs[ctx->nsegs - 1];
759 }
760 } else {
761 seg = &ctx->segs[idx];
762 }
763
764 int32_t uV = seg->slope_uV_per_LSB * (int32_t)code + seg->offset_uV;
765 /* Round to nearest mV */
766 return (uV >= 0) ? (uV + 500) / 1000 : (uV - 500) / 1000;
767}
768
769int32_t wise_asaradc_code_to_mV(uint16_t code)
770{
771 return asaradc_code_to_mV(&g_calib_ctx, code);
772}
