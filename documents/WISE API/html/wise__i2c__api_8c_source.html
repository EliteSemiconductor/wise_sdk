32static void wise_i2c_slave_recover(uint8_t i2cIntf, WISE_I2C_CTRL_T *i2c)
33{
34 /*
35 * The entire recovery (HW reconfigure + context setup + arm) must be
36 * atomic w.r.t. the I2C ISR. Without this, the config path enables
37 * IICEn + NVIC before arm_rx sets up the context — a master transaction
38 * arriving in that window triggers F_AH preload, but then arm_rx's
39 * CLEAR_FIFO wipes the data mid-transfer → slave returns 00 00 00 00.
40 *
41 * __disable_irq() (CPSID) globally masks all interrupts. arm_rx
42 * internally calls __enable_irq() after the context is fully set up,
43 * so interrupts are re-enabled only when the slave is ready.
44 */
45 __disable_irq();
46
47 hal_intf_i2c_config(i2cIntf, &i2c->cfg);
48
49 if (i2c->slave_rx_buf && i2c->slave_rx_len > 0) {
50 wise_i2c_slave_arm_rx(i2cIntf, i2c->slave_rx_buf, i2c->slave_rx_len);
51 } else {
52 __enable_irq();
53 }
54}

156/*
157 * Internal callback wrapper for slave watchdog.
158 * Intercepts ADDRESS_HIT / TRANSFER_DONE to track bus-active state,
159 * then forwards the event to the user's callback.
160 */
161static void _i2c_slave_watchdog_cb(uint8_t ch, uint32_t event, void *ctx)
162{
163 (void)ctx;
164 WISE_I2C_CTRL_T *i2c = &i2c_ctrl_array[ch];
165
166 if (event == WISE_I2C_EVENT_ADDRESS_HIT) {
167 i2c->slave_bus_active = 1;
168 i2c->slave_addr_hit_tick = wise_tick_get_counter();
169 }
170 if (event == WISE_I2C_EVENT_TRANSFER_DONE) {
171 i2c->slave_bus_active = 0;
172 i2c->slave_last_done_tick = wise_tick_get_counter();
173 }
174
175 if (i2c->user_cb) {
176 i2c->user_cb(ch, event, i2c->user_ctx);
177 }
178}

202static inline WISE_STATUS _wise_i2c_xfer_core(uint8_t i2cIntf, bool set_addr, WISE_I2C_MSG_T *i2c_msg)
203
204{
205 if (i2c_msg->buf_ptr == NULL) {
206 return WISE_INVALID_INDEX;
207 }
208 if (i2c_msg->data_count > CHIP_I2C_MAX_XFER) {
209 return WISE_INVALID_INDEX;
210 }
211
212 WISE_I2C_CTRL_T *i2c = &i2c_ctrl_array[i2cIntf];
213 if (set_addr) {
214 i2c->cfg.target_address = i2c_msg->target_address;
215 }
216
217 HAL_I2C_MSG_T hal_i2c_msg = {
218 .en_stop = i2c_msg->en_stop, .en_data = i2c_msg->en_data, .en_addr = i2c_msg->en_addr, .en_start = i2c_msg->en_start};
219
220 HAL_STATUS ret;
221 if (i2c_msg->dir == WISE_I2C_DIR_TX) {
222 ret = hal_intf_i2c_transmit_ex(i2cIntf, &i2c->cfg, &hal_i2c_msg, i2c_msg->buf_ptr, i2c_msg->data_count);
223 } else {
224 ret = hal_intf_i2c_receive_ex(i2cIntf, &i2c->cfg, &hal_i2c_msg, i2c_msg->buf_ptr, i2c_msg->data_count);
225 }
226
227 return (ret == HAL_NO_ERR) ? WISE_SUCCESS : WISE_FAIL;
228}

313int32_t wise_i2c_transfer(uint8_t intf, const WISE_I2C_XFER_MSG_T *msgs, size_t nmsgs, uint32_t timeout_ms)
314{
315 if (!msgs || nmsgs == 0) {
316 return WISE_FAIL;
317 }
318 if (!IS_VALID_I2C_CHANNEL(intf)) {
319 return WISE_INVALID_INDEX;
320 }
321
322 for (size_t i = 0; i < nmsgs; ++i) {
323 if (!msgs[i].buf || msgs[i].len == 0) {
324 return WISE_INVALID_INDEX;
325 }
326 if (msgs[i].len > CHIP_I2C_MAX_XFER) {
327 return WISE_INVALID_INDEX;
328 }
329 }
330
331 WISE_I2C_CTRL_T *i2c = &i2c_ctrl_array[intf];
332 uint8_t saved_addressing = i2c->cfg.addressing;
333
334 WISE_STATUS st = WISE_SUCCESS;
335
336 for (size_t i = 0; i < nmsgs; ++i) {
337 const WISE_I2C_XFER_MSG_T *m = &msgs[i];
338
339 i2c->cfg.addressing = (m->flags & WISE_I2C_M_TEN) ? I2C_ADDR_10_BITS : I2C_ADDR_7_BITS;
340
341 bool is_read = (m->flags & WISE_I2C_M_RD) != 0;
342 bool is_last = (i == (nmsgs - 1));
343 bool no_stop = (!is_last) || (m->flags & WISE_I2C_M_NO_STOP);
344
345 WISE_I2C_MSG_T low = {.buf_ptr = m->buf,
346 .data_count = m->len,
347 .dir = is_read ? WISE_I2C_DIR_RX : WISE_I2C_DIR_TX,
348 .target_address = m->addr,
349 .en_start = ENABLE,
350 .en_addr = ENABLE,
351 .en_data = ENABLE,
352 .en_stop = no_stop ? DISABLE : ENABLE};
353
354 st = _wise_i2c_xfer_core(intf, /*set_addr=*/true, &low);
355 if (st != WISE_SUCCESS) {
356 break;
357 }
358
359 /*
360 * If timeout_ms > 0, wait for segment completion before proceeding
361 * to the next message. This ensures proper repeated-start semantics
362 * for combined transactions (e.g. mem_read: write reg + read data).
363 *
364 * If timeout_ms == 0, fire-and-forget (caller handles completion,
365 * e.g. wise_i2c_probe does its own polling loop).
366 */
367 if (timeout_ms > 0) {
368 bool done = false, ack = false, arb_lost = false;
369 uint32_t elapsed;
370 for (elapsed = 0; elapsed <= timeout_ms; ++elapsed) {
371 if (hal_intf_i2c_get_probe_state(intf, &done, &ack, &arb_lost) != HAL_NO_ERR) {
372 st = WISE_FAIL;
373 break;
374 }
375 if (arb_lost) {
376 st = WISE_FAIL;
377 break;
378 }
379 if (done) {
380 break;
381 }
382 wise_tick_delay_ms(1);
383 }
384 if (st != WISE_SUCCESS) {
385 break;
386 }
387 if (!done) {
388 st = WISE_FAIL; /* timeout */
389 break;
390 }
391 }
392 }
393
394 i2c->cfg.addressing = saved_addressing;
395
396 return (st == WISE_SUCCESS) ? WISE_SUCCESS : WISE_FAIL;
397}

399int32_t wise_i2c_probe(uint8_t intf, uint16_t addr, uint32_t timeout_ms)
400{
401 bool done = false;
402 bool ack = false;
403 bool arb_lost = false;
404
405 if (!IS_VALID_I2C_CHANNEL(intf)) {
406 return WISE_INVALID_INDEX;
407 }
408
409 uint8_t dummy = 0;
410 WISE_I2C_XFER_MSG_T msg = {
411 .addr = addr,
412 .flags = 0,
413 .buf = &dummy,
414 .len = 1,
415 };
416
417 if (wise_i2c_transfer(intf, &msg, 1, 0) != WISE_SUCCESS) {
418 return WISE_FAIL;
419 }
420
421 for (uint32_t elapsed = 0; elapsed <= timeout_ms; ++elapsed) {
422 if (hal_intf_i2c_get_probe_state(intf, &done, &ack, &arb_lost) != HAL_NO_ERR) {
423 return WISE_FAIL;
424 }
425 if (arb_lost) {
426 return WISE_FAIL;
427 }
428 if (done) {
429 return ack ? WISE_SUCCESS : WISE_FAIL;
430 }
431 wise_tick_delay_ms(1);
432 }
433
434 return WISE_FAIL;
435}

437int32_t wise_i2c_mem_read(uint8_t intf, uint16_t addr, uint32_t reg, uint8_t reg_len, uint8_t *rx, uint16_t count, uint32_t timeout_ms)
438{
439 if (!rx || !count || reg_len == 0 || reg_len > 4) {
440 return WISE_INVALID_INDEX;
441 }
442
443 uint8_t hdr[4];
444 /* MSB-first */
445 for (uint8_t i = 0; i < reg_len; ++i) {
446 hdr[i] = (uint8_t)((reg >> (8 * (reg_len - 1 - i))) & 0xFF);
447 }
448
449 WISE_I2C_XFER_MSG_T seq[2] = {
450 {.addr = addr, .flags = 0, .buf = hdr, .len = reg_len}, /* SLA+W + reg bytes */
451 {.addr = addr, .flags = WISE_I2C_M_RD, .buf = rx, .len = count} /* RESTART + SLA+R + read */
452 };
453 return wise_i2c_transfer(intf, seq, 2, timeout_ms);
454}

463WISE_STATUS wise_i2c_slave_arm_rx(uint8_t i2cIntf, uint8_t *buf, uint32_t len)
464{
465 if (IS_VALID_I2C_CHANNEL(i2cIntf)) {
466 i2c_ctrl_array[i2cIntf].slave_rx_buf = buf;
467 i2c_ctrl_array[i2cIntf].slave_rx_len = len;
468 i2c_ctrl_array[i2cIntf].slave_armed = 1;
469 /* Do NOT set slave_last_done_tick here — idle watchdog should only
470 * trigger after a transaction has started (ADDRESS_HIT) but never
471 * completed. Setting it on arm causes false timeouts during idle. */
472 }
473 /*
474 * Do NOT touch hardware (Dir, FIFO) here — the IRQ from the previous
475 * transfer may still be enabled. All hardware setup is done atomically
476 * inside hal_drv_i2c_transfer → i2c_transfer_er8130, which disables
477 * the NVIC IRQ first, clears status/FIFO, then re-enables.
478 */
479 return wise_i2c_recv_nbyte(i2cIntf, buf, len);
480}

487int32_t wise_i2c_mem_write(uint8_t intf, uint16_t addr, uint32_t reg, uint8_t reg_len, const uint8_t *tx, uint16_t count, uint32_t timeout_ms)
488{
489 if (!tx || !count || reg_len == 0 || reg_len > 4) {
490 return WISE_INVALID_INDEX;
491 }
492
493 uint8_t tmp[4 + 255];
494 if ((uint32_t)reg_len + count > sizeof(tmp)) {
495 return WISE_INVALID_INDEX;
496 }
497
498 for (uint8_t i = 0; i < reg_len; ++i) {
499 tmp[i] = (uint8_t)((reg >> (8 * (reg_len - 1 - i))) & 0xFF);
500 }
501 memcpy(&tmp[reg_len], tx, count);
502
503 WISE_I2C_XFER_MSG_T m = {.addr = addr, .flags = 0, .buf = tmp, .len = (uint16_t)(reg_len + count)};
504 return wise_i2c_transfer(intf, &m, 1, timeout_ms);
505}

514int32_t wise_i2c_slave_proc(uint8_t i2cIntf)
515{
516 if (!IS_VALID_I2C_CHANNEL(i2cIntf)) {
517 return 0;
518 }
519
520 WISE_I2C_CTRL_T *i2c = &i2c_ctrl_array[i2cIntf];
521
522 if (i2c->slave_timeout_ms == 0) {
523 return 0;
524 }
525
526 uint32_t now = wise_tick_get_counter();
527 bool need_recovery = false;
528
529 if (i2c->slave_bus_active) {
530 /* ADDRESS_HIT fired but TRANSFER_DONE never came */
531 if ((now - i2c->slave_addr_hit_tick) >= MS_TO_CLK(i2c->slave_timeout_ms)) {
532 need_recovery = true;
533 }
534 }
535
536 /*
537 * Idle watchdog: if slave completed a transaction (last_done_tick != 0)
538 * but hasn't seen another TRANSFER_DONE for 2×timeout, the I2C
539 * peripheral may be in a bad state. Force full re-initialization.
540 * Note: last_done_tick is only set on TRANSFER_DONE, NOT on arm,
541 * so this won't trigger during normal idle waiting.
542 */
543 if (!need_recovery && i2c->slave_armed && !i2c->slave_bus_active
544 && i2c->slave_last_done_tick != 0) {
545 if ((now - i2c->slave_last_done_tick) >= MS_TO_CLK(i2c->slave_timeout_ms * 2)) {
546 need_recovery = true;
547 }
548 }
549
550 if (!need_recovery) {
551 return 0;
552 }
553
554 /* Timeout — reconfigure hardware and re-arm slave RX */
555 i2c->slave_bus_active = 0;
556 i2c->slave_armed = 0;
557 i2c->slave_last_done_tick = 0; /* prevent repeated idle-timeout recovery */
558 wise_i2c_slave_recover(i2cIntf, i2c);
559
560 return 1;
561}
