65{

88static void _wise_uart_rx_isr(void *context, uint8_t uart_idx)
89{
90 ST_WISE_UART_CTRL_T *uCtrl = (ST_WISE_UART_CTRL_T *)&uartCtrl[uart_idx];
91 uint8_t rByte = 0;
92
93 if (uCtrl->rxFifo) {
94 if (uartCtrl[uart_idx].dma_enable != ENABLE) {
95 while (WISE_SUCCESS == hal_intf_uart_poll_byte(uart_idx, &rByte)) {
96 wise_fifo_push_char(uCtrl->rxFifo, rByte);
97 }
98 }
99#if (defined CHIP_HAS_DMA) && (defined CHIP_DMA_SUPPORT_PERIPHERAL)
100 else {
101 uint8_t *dma_rx = &uart_dma_rx_byte[uart_idx];
102
103 if (hal_intf_dma_channel_setup(HAL_DMA_UART_RX(uart_idx), NULL, dma_rx, 1, 0, 0) == HAL_ERR) {
104 return;
105 }
106 hal_intf_dma_channel_trigger(HAL_DMA_UART_RX(uart_idx));
107 wise_fifo_push_char(uCtrl->rxFifo, *dma_rx);
108 }
109#endif
110 } else if (uCtrl->rxCallback) {
111 while (WISE_SUCCESS == hal_intf_uart_poll_byte(uart_idx, &rByte)) {
112 (uCtrl->rxCallback)(uart_idx, rByte);
113 }
114 }
115}

179void wise_uart_enable(uint8_t uartIntf, uint8_t opMode)
180{
181 if (uartIntf >= CHIP_UART_CHANNEL_NUM) {
182 return;
183 }
184
185 if (opMode == WISE_UART_FLAG_NONE) {
186 hal_intf_uart_disable_irq(uartIntf);
187 uartCtrl[uartIntf].onFlag = opMode;
188
189 hal_intf_module_clk_disable(uartCtrl[uartIntf].clockSel);
190
191 return;
192 }
193
194 uartCtrl[uartIntf].onFlag = opMode;
195
196 if (uartCtrl[uartIntf].rxFifo) {
197 wise_fifo_reset(uartCtrl[uartIntf].rxFifo);
198 }
199
200 if (uartCtrl[uartIntf].txFifo) {
201 wise_fifo_reset(uartCtrl[uartIntf].txFifo);
202 }
203
204 //reset uart HW fifo & int status
205 hal_intf_uart_reset_fifo(uartIntf, HAL_UART_RESET_ALL);
206 if (uartIntf == 0) {
207 NVIC_ClearPendingIRQ(UART0_IRQn);
208 } else if (uartIntf == 1) {
209 NVIC_ClearPendingIRQ(UART1_IRQn);
210 } else if (uartIntf == 2) {
211 NVIC_ClearPendingIRQ(UART2_IRQn);
212 }
213
214 if ((uartCtrl[uartIntf].onFlag & WISE_UART_FLAG_RX) && (uartCtrl[uartIntf].rxFifo != NULL)) {
215 hal_intf_uart_enable_irq(uartIntf, HAL_UART_EVT_RX_FIN);
216 }
217
218 if ((uartCtrl[uartIntf].onFlag & WISE_UART_FLAG_TX) && (uartCtrl[uartIntf].txFifo != NULL)) {
219 hal_intf_uart_enable_irq(uartIntf, HAL_UART_EVT_TX_FIN);
220 }
221
222 // TODO: TX FIFO?
223}

358static void _wise_uart_process_tx_fifo(uint8_t channel)
359{
360 CORE_DECLARE_IRQ_STATE;
361
362 if (!(uartCtrl[channel].txFifo)) {
363 return;
364 }
365
366 if (!CORE_IS_IN_ISR()) {
367 CORE_ENTER_CRITICAL();
368 }
369
370 if (!UART_IS_BUSY(channel)) {
371 uint8_t outputC;
372
373 if (WISE_SUCCESS == wise_fifo_pop_char_no_crit(uartCtrl[channel].txFifo, &outputC)) {
374 if (uartCtrl[channel].dma_enable != ENABLE) {
375 hal_intf_uart_write_data(0, &outputC, 1);
376 UART_SET_BUSY(channel);
377 } else {
378#if (defined CHIP_HAS_DMA) && (defined CHIP_DMA_SUPPORT_PERIPHERAL)
379 uint8_t *dma_tx = &uart_dma_tx_byte[channel];
380 *dma_tx = outputC;
381 if (hal_intf_dma_channel_setup(HAL_DMA_UART_TX(channel), dma_tx, NULL, 1, 0, 1) == HAL_ERR) {
382 return; //FAIL
383 }
384 hal_intf_dma_channel_trigger(HAL_DMA_UART_TX(channel));
385#endif
386 }
387 }
388 }
389
390 if (!CORE_IS_IN_ISR()) {
391 CORE_EXIT_CRITICAL();
392 }
393}
